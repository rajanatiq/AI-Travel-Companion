IF OBJECT_ID('dbo.users', 'U') IS NULL
BEGIN
    CREATE TABLE dbo.users (
        id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
        email NVARCHAR(255) NOT NULL,
        password_hash NVARCHAR(255) NOT NULL,
        home_locale NVARCHAR(10) NOT NULL DEFAULT 'en',
        created_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
        updated_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),

-- 2. PREFERENCES TABLE (1-to-1 with users)
-------------------------------------------------------------------------------
IF OBJECT_ID('dbo.preferences', 'U') IS NULL
BEGIN
    CREATE TABLE dbo.preferences (
        user_id UNIQUEIDENTIFIER PRIMARY KEY,
        interests NVARCHAR(MAX) NOT NULL DEFAULT '[]', -- JSON Array format: '["history", "food", "art"]'
        pace_preference NVARCHAR(20) NOT NULL DEFAULT 'relaxed',
        default_budget_tier INT NOT NULL DEFAULT 2,
        updated_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),

        CONSTRAINT fk_preferences_users FOREIGN KEY (user_id) 
            REFERENCES dbo.users(id) ON DELETE CASCADE,
        CONSTRAINT chk_preferences_pace CHECK (pace_preference IN ('relaxed', 'balanced', 'packed')),
        CONSTRAINT chk_preferences_budget CHECK (default_budget_tier BETWEEN 1 AND 4),
        CONSTRAINT chk_preferences_interests_json CHECK (ISJSON(interests) = 1)
    );
END
GO

-- 3. TRIPS TABLE
-------------------------------------------------------------------------------
IF OBJECT_ID('dbo.trips', 'U') IS NULL
BEGIN
    CREATE TABLE dbo.trips (
        id UNIQUEIDENTIFIER PRIMARY KEY DEFAULT NEWID(),
        user_id UNIQUEIDENTIFIER NOT NULL,
        destination NVARCHAR(255) NOT NULL,
        start_date DATE NOT NULL,
        end_date DATE NOT NULL,
        budget_total DECIMAL(10, 2) NOT NULL,
        interests NVARCHAR(MAX) NOT NULL DEFAULT '[]', -- JSON Array format
        pace NVARCHAR(20) NOT NULL DEFAULT 'balanced',
        status NVARCHAR(20) NOT NULL DEFAULT 'draft',
        created_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
        updated_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),

        CONSTRAINT fk_trips_users FOREIGN KEY (user_id) 
            REFERENCES dbo.users(id) ON DELETE CASCADE,
        CONSTRAINT chk_trips_dates CHECK (end_date >= start_date),
        CONSTRAINT chk_trips_budget CHECK (budget_total > 0),
        CONSTRAINT chk_trips_status CHECK (status IN ('draft', 'confirmed', 'completed', 'archived')),
        CONSTRAINT chk_trips_pace CHECK (pace IN ('relaxed', 'balanced', 'packed')),
        CONSTRAINT chk_trips_interests_json CHECK (ISJSON(interests) = 1)
    );

    CREATE NONCLUSTERED INDEX idx_trips_user_id ON dbo.trips(user_id);
    CREATE NONCLUSTERED INDEX idx_trips_start_date ON dbo.trips(start_date DESC);
    CREATE NONCLUSTERED INDEX idx_trips_status ON dbo.trips(status);
    CREATE NONCLUSTERED INDEX idx_trips_user_created ON dbo.trips(user_id, created_at DESC);
END
GO

        user_edited BIT NOT NULL DEFAULT 0,
        created_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),
        updated_at DATETIMEOFFSET NOT NULL DEFAULT SYSDATETIMEOFFSET(),

        CONSTRAINT fk_itinerary_trips FOREIGN KEY (trip_id) 
            REFERENCES dbo.trips(id) ON DELETE CASCADE,
        CONSTRAINT uq_itinerary_trip_day_order UNIQUE (trip_id, day_index, order_index),
        CONSTRAINT chk_itinerary_day CHECK (day_index >= 0),
        CONSTRAINT chk_itinerary_order CHECK (order_index >= 0),
        CONSTRAINT chk_itinerary_duration CHECK (duration_min > 0),
        CONSTRAINT chk_itinerary_cost CHECK (est_cost IS NULL OR est_cost > 0),
        CONSTRAINT chk_itinerary_crowd CHECK (crowd_level IS NULL OR crowd_level IN ('low', 'medium', 'high')),
        CONSTRAINT chk_itinerary_category CHECK (category IN (
            'history', 'food', 'nature', 'nightlife', 'art', 'adventure', 
            'family', 'shopping', 'culture', 'meal', 'transport'
        )),
        CONSTRAINT chk_itinerary_opening_hours_json CHECK (opening_hours IS NULL OR ISJSON(opening_hours) = 1)
    );

    CREATE NONCLUSTERED INDEX idx_itinerary_trip ON dbo.itinerary_items(trip_id);
    CREATE NONCLUSTERED INDEX idx_itinerary_day ON dbo.itinerary_items(trip_id, day_index);
    
    -- Filtered Index for items edited by user
    CREATE NONCLUSTERED INDEX idx_itinerary_user_edited ON dbo.itinerary_items(trip_id) 
        WHERE user_edited = 1;

    -- Spatial Index for Geospatial Distance Queries
    CREATE SPATIAL INDEX sx_itinerary_geo ON dbo.itinerary_items(geo_location);

-------------------------------------------------------------------------------
-- Triggers for Geo-spatial location calculation (Lat/Lon -> Geography Point)
-------------------------------------------------------------------------------
CREATE OR ALTER TRIGGER trg_itinerary_items_geography
ON dbo.itinerary_items
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    IF UPDATE(lat) OR UPDATE(lon)
    BEGIN
        UPDATE t
        SET geo_location = CASE 
            WHEN i.lat IS NOT NULL AND i.lon IS NOT NULL 
            THEN geography::Point(i.lat, i.lon, 4326)
            ELSE NULL 
        END
        FROM dbo.itinerary_items t
        INNER JOIN inserted i ON t.id = i.id;
    END
END;
GO

CREATE OR ALTER TRIGGER trg_places_cache_geography
ON dbo.places_cache
AFTER INSERT, UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    IF UPDATE(lat) OR UPDATE(lon)
    BEGIN
        UPDATE p
        SET geo_location = CASE 
            WHEN i.lat IS NOT NULL AND i.lon IS NOT NULL 
            THEN geography::Point(i.lat, i.lon, 4326)
            ELSE NULL 
        END
        FROM dbo.places_cache p
        INNER JOIN inserted i ON p.place_id = i.place_id;
    END
END;
GO

-- Triggers for `updated_at` timestamp management
-------------------------------------------------------------------------------
CREATE OR ALTER TRIGGER trg_users_updated_at
ON dbo.users
AFTER UPDATE
AS
BEGIN
    SET NOCOUNT ON;
    IF NOT UPDATE(updated_at)
    BEGIN
        UPDATE u
        SET updated_at = SYSDATETIMEOFFSET()
        FROM dbo.users u
        INNER JOIN inserted i ON u.id = i.id;
    END
END;
